scripts/full_pipeline.py	                                      📜 主流水线脚本	~10KB
E2E/i3d.py	                                                  I3D 模型定义
configs/id3_i3d.yaml	                                          TriDet 模型配置（20类, 2048-dim）
epoch_039.pth.tar	                                              TriDet 训练权重（gitignored）	（训练得到）
E2E/model/rgb_imagenet.pt	                                  I3D RGB 预训练权重	（自行下载）
E2E/model/flow_imagenet.pt	                                  I3D Flow 预训练权重	（自行下载）
./data/id3_2048/annotations.json	                          标签定义（20类 THUMOS14 标签）
scripts/extract_features.py                                       单独启用 I3D 特征提取

输入格式   直接由视频输出结果                                        以 I3D 特征直接输出结果
scripts/full_pipeline.py 
--video 视频路径
--config configs/id3_i3d.yaml 
--ckpt epoch_039.pth.tar 
--output_dir 结果输出                                              结果会以 txt 形式输出
# 从原始视频提取 RGB 特征 (2048-dim)
    python scripts/extract_features.py \
    --video_path 视频路径
    --output_dir 输出路径
    --mode rgb+flow \
    --device cpu

# 从预提取帧提取 RGB+Flow 特征 (2048-dim)                           请注意，以帧形式输出 I3D 特征时，至少需要16帧画面
python scripts/extract_features.py \
    --frames_dir 帧路径
    --output_dir 输出路径
    --mode rgb+flow \
    --device cpu
